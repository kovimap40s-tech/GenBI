Koneksi Langsung Folder Google Drive

Berfungsi 100% di file lokal (file://) tanpa error OAuth 404

Kenapa muncul Error 404 Google sebelumnya? Karena file HTML dibuka langsung dari komputer (file:///...), sehingga Google memblokir popup OAuth. Dengan Google Apps Script Web App (gratis & milik akun Google Anda sendiri), siapa pun bisa langsung mengunggah file ke folder 1B07jomfmZEPWigZt8JzNKeC0eoHIFJ6J tanpa popup login!

URL disimpan di Firebase sehingga otomatis aktif untuk semua anggota tim Anda.

Langkah Cepat (1 Menit Selesai):
  1. Buka editor proyek Google Apps Script Anda dengan akun pemilik folder Google Drive.
  2. Ganti kode dengan kode di bawah ini, pilih fungsi izinDrive di toolbar atas, lalu klik Run (▶) → Review permissions → Allow.
  3. Klik Deploy → Manage deployments → Edit (ikon pensil) → Version: New version → Deploy (agar izin `DriveApp` aktif di Web App).
function izinDrive() {
  // Pilih fungsi ini lalu klik Run (▶) sekali untuk menyetujui izin Google Drive
  var folder = DriveApp.getFolderById("1B07jomfmZEPWigZt8JzNKeC0eoHIFJ6J");
  Logger.log("Berhasil terhubung ke folder: " + folder.getName());
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var folder = DriveApp.getFolderById("1B07jomfmZEPWigZt8JzNKeC0eoHIFJ6J");
    var decoded = Utilities.base64Decode(data.base64);
    var blob = Utilities.newBlob(decoded, data.mimeType || "image/jpeg", data.fileName);
    var file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return ContentService.createTextOutput(JSON.stringify({
      status: "ok",
      fileId: file.getId(),
      driveUrl: "https://drive.google.com/file/d/" + file.getId() + "/view?usp=sharing"
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
